def _json_default(o):
    # Make numpy types JSON serializable
    if isinstance(o, (np.integer,)):
        return int(o)
    if isinstance(o, (np.floating,)):
        return float(o)
    if isinstance(o, (np.ndarray,)):
        return o.tolist()
    return str(o)

def _safe_git_info():
    try:
        commit = subprocess.check_output(["git", "rev-parse", "HEAD"], stderr=subprocess.STDOUT).decode().strip()
    except Exception:
        commit = None
    try:
        status = subprocess.check_output(["git", "status", "--porcelain"], stderr=subprocess.STDOUT).decode().strip()
        dirty = bool(status)
    except Exception:
        dirty = None
    return {"git_commit": commit, "git_dirty": dirty}

def _save_json(path: Path, obj: dict):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, indent=2, default=_json_default)

def _try_stack_ersps(ersp_list):
    """
    Try to stack list of (nF,nT) into (n,nF,nT). If shapes differ, return None.
    """
    if not isinstance(ersp_list, (list, tuple)) or len(ersp_list) == 0:
        return None
    shapes = [np.asarray(e).shape for e in ersp_list]
    if len(set(shapes)) != 1:
        return None
    return np.stack([np.asarray(e, dtype=np.float32) for e in ersp_list], axis=0)

import numpy as np
from pathlib import Path

def load_ersp_keep_from_run(run_dir: Path):
    f_npy = Path(run_dir) / "ersp_keep.npy"
    f_npz = Path(run_dir) / "ersp_keep.npz"

    if f_npy.exists():
        arr = np.load(f_npy, allow_pickle=False)  # expected (n, nF, nT)
        ersp_keep = [arr[i] for i in range(arr.shape[0])]
        return ersp_keep, {"format": "npy_stacked", "shape": arr.shape}

    if f_npz.exists():
        z = np.load(f_npz, allow_pickle=False)
        # keys are ersp_000000, ersp_000001, ... -> sort to preserve sample order
        keys = sorted(z.files)
        ersp_keep = [z[k] for k in keys]
        return ersp_keep, {"format": "npz_per_sample", "n_samples": len(keys)}

    return None, {"format": None}

# ------------------------------------------------------------
# Step 1) Build "-101" segmentation maps
#   +1 where ERSP >= thr_pos
#   -1 where ERSP <= thr_neg
#    0 otherwise
# ------------------------------------------------------------
def ersp_to_minus101(ersp, thr_pos, thr_neg):
    pos = (ersp >= thr_pos)
    neg = (ersp <= thr_neg)
    seg = np.zeros_like(ersp, dtype=np.int8)
    seg[pos] = 1
    seg[neg] = -1
    return seg

# ------------------------------------------------------------
# Step 2) Extract geometry-only blob descriptors from segmentation
# Each blob described by normalized geometry features (no intensity):
#   sign ∈ {+1,-1}
#   t_center, f_center  (centroid, normalized to [0,1])
#   t_span, f_span      (extent, normalized)
#   area_frac           (#pixels / total pixels)
# ------------------------------------------------------------
def extract_blobs_minus101(seg):
    nF, nT = seg.shape
    blobs = []

    for sign in (1, -1):
        mask = (seg == sign)
        lab, nlab = label(mask, structure=STRUCT)
        if nlab == 0:
            continue

        for lab_id in range(1, nlab + 1):
            coords = np.argwhere(lab == lab_id)   # rows are [f, t]
            if coords.size == 0:
                continue

            f = coords[:, 0].astype(np.float32)
            t = coords[:, 1].astype(np.float32)

            # Geometry
            f0, f1 = float(f.min()), float(f.max())
            t0, t1 = float(t.min()), float(t.max())

            f_center = float(f.mean())
            t_center = float(t.mean())

            f_span = (f1 - f0 + 1.0)
            t_span = (t1 - t0 + 1.0)

            area = float(coords.shape[0])

            # Normalize to be invariant to n_freq/n_time
            blob = {
                "sign": int(sign),
                "t_center": t_center / max(1.0, (nT - 1.0)),
                "f_center": f_center / max(1.0, (nF - 1.0)),
                "t_span":   t_span   / max(1.0, nT),
                "f_span":   f_span   / max(1.0, nF),
                "area_frac": area / float(nF * nT),
            }
            blobs.append(blob)

    # Rank blobs by area (largest first); keep top MAXB
    blobs.sort(key=lambda b: b["area_frac"], reverse=True)
    return blobs[:MAXB]

# ------------------------------------------------------------
# Step 3) Blob-to-blob distance (geometry + sign)
# We intentionally ignore intensity; this is strictly "-101" geometry.
# ------------------------------------------------------------
def blob_dist(b1, b2, w=None, sign_mismatch_penalty=0.5):
    # Feature vector order
    v1 = np.array([b1["t_center"], b1["f_center"], b1["t_span"], b1["f_span"], b1["area_frac"]], dtype=np.float32)
    v2 = np.array([b2["t_center"], b2["f_center"], b2["t_span"], b2["f_span"], b2["area_frac"]], dtype=np.float32)

    if w is None:
        w = np.ones_like(v1)

    d = (v1 - v2) * w
    base = float(np.sqrt(np.sum(d * d)))

    # Penalize matching a positive blob to a negative blob (optional, but usually desirable)
    if b1["sign"] != b2["sign"]:
        base += float(sign_mismatch_penalty)

    return base

# ------------------------------------------------------------
# Step 4) Sample-to-sample distance via optimal assignment (Hungarian)
# Here we do a simple min-cost matching between blob sets.
# Unmatched blobs get a penalty (because "extra blobs" should matter).
# ------------------------------------------------------------
from scipy.optimize import linear_sum_assignment

def sample_dist(blobsA, blobsB, w=None, unmatched_penalty=0.5, sign_mismatch_penalty=0.5, normalize_by="max"):
    nA, nB = len(blobsA), len(blobsB)

    if nA == 0 and nB == 0:
        return 0.0
    if nA == 0 or nB == 0:
        base = unmatched_penalty * abs(nA - nB)
        denom = max(nA, nB) if normalize_by == "max" else max(1, int(np.round((nA + nB) / 2.0)))
        return float(base / denom)

    # Cost matrix
    C = np.zeros((nA, nB), dtype=np.float32)
    for i in range(nA):
        for j in range(nB):
            C[i, j] = blob_dist(blobsA[i], blobsB[j], w=w, sign_mismatch_penalty=sign_mismatch_penalty)

    # Optimal assignment
    ri, ci = linear_sum_assignment(C)
    match_cost = float(C[ri, ci].sum())

    # Unmatched penalty
    unmatched = abs(nA - nB)
    total = match_cost + unmatched_penalty * unmatched

    # Normalize
    if normalize_by == "max":
        denom = max(nA, nB)
    elif normalize_by == "mean":
        denom = max(1, int(np.round((nA + nB) / 2.0)))
    else:
        denom = 1
    return float(total / denom)

def compute_medoid(D, idx):
    Dc = D[np.ix_(idx, idx)]
    mean_dist = Dc.mean(axis=1)
    j = int(np.argmin(mean_dist))
    return int(idx[j]), float(mean_dist[j])

def ersp_to_minus101(ersp, thr_pos, thr_neg):
    seg = np.zeros_like(ersp, dtype=np.int8)
    seg[ersp >= thr_pos] = 1
    seg[ersp <= thr_neg] = -1
    return seg

def compute_cluster_medoids(D, labels):
    clusters = np.unique(labels)
    medoid_by_cluster = {}
    medoid_mean_intra = {}
    for c in clusters:
        idx = np.where(labels == c)[0]
        if len(idx) == 0:
            continue
        Dc = D[np.ix_(idx, idx)]
        mean_dist = Dc.mean(axis=1)
        j = int(np.argmin(mean_dist))
        medoid_idx = int(idx[j])
        medoid_by_cluster[int(c)] = medoid_idx
        medoid_mean_intra[int(c)] = float(mean_dist[j])
    return medoid_by_cluster, medoid_mean_intra
